#+begin_src python
class Point:
    def __init__(self, x, y): self.x, self.y = x, y
    def _repr_html_(self): return f"<b>Point</b>({self.x}, {self.y})"
#+end_src

*** cl-stack
#+begin_src lisp :exports source :results silent
(asdf:load-system "canvas-backend-svg")

(defclass point () ((x :initarg :x) (y :initarg :y)))
#+end_src

#+begin_src lisp :post shorten(*this*)
(defmethod canvas:render ((p point))
  (canvas:el :p nil (canvas:el :b nil "Point") (format nil "(~a, ~a)" (slot-value p 'x) (slot-value p 'y))))

(canvas:to-html (list (make-instance 'point :x 1 :y 2)
                      (canvas:make-chart :scatter :title "Points"
                                         :series (list (canvas:make-series "p" '((1 . 2) (3 . 1)))))))
#+end_src

LangChain is a framework for building agents and LLM-powered applications.

** Features
:PROPERTIES:
:UNNUMBERED: t
:END:
+ LangChain / LangGraph: agents with tools, memory, graphs of steps, provider integrations
+ LlamaIndex: loaders, chunking, vector indexes, retrievers, query engines (RAG)
+ MCP SDK / FastMCP: servers and clients for tools, resources, prompts
+ OpenAI Agents SDK, PydanticAI: agent loops, handoffs, guardrails, tracing

#+begin_src lisp :exports source :results silent
(load "examples/mock-llm.lisp")
(mock-llm:start-mock)

(asdf:load-system "llm-protocol-openai")
(asdf:load-system "ai-agent-protocol/mcp")
(asdf:load-system "rpc-backend-inprocess")
(asdf:load-system "rag-backend-hybrid")
(asdf:load-system "rag-backend-memory")
(asdf:load-system "rag-backend-text")
(asdf:load-system "http-backend-async")
(asdf:load-system "event-backend-libuv")

(setf http-backend-async:*event-backend-maker* #'event-backend-libuv:make-libuv-backend
      http-protocol:*http-backend* (http-backend-async:make-async-backend))

(defvar *llm* (stack-llm-openai:make-openai-compat-backend
               :base-url "http://127.0.0.1:18080/v1" :api-key "sk-test" :default-model "gpt-4o-mini"))

(defmacro with-agent-loop (&body body)
  "Run BODY with a libuv event loop bound; agents await on it."
  `(let* ((eb (event-backend-libuv:make-libuv-backend))
          (el (event-protocol:make-event-loop eb)))
     (unwind-protect
          (event-protocol:with-event-backend (eb)
            (event-protocol:with-event-loop-var (el) ,@body))
       (event-protocol:close-event-loop eb el))))

(defparameter *city-schema*
  (com.inuoe.jzon:parse "{\"type\":\"object\",\"properties\":{\"city\":{\"type\":\"string\"}},\"required\":[\"city\"]}"))
#+end_src

An agent with a tool:
#+begin_src python
@tool
def get_weather(city: str) -> str:
    return "18°C, cloudy"

agent = create_agent(model, tools=[get_weather], system_prompt="Answer with the tool.")
agent.invoke({"messages": [("user", "What's the weather in Paris?")]})
#+end_src

#+begin_src lisp
(let ((calls '())
      (agent (stack-ai-agent:make-ai-agent :name "weather" :backend *llm*
                                           :instructions "Answer with the tool.")))
  (stack-ai-agent:register-agent-tool
   agent (stack-ai-agent:make-function-tool
          :name "get_weather" :description "Current weather for a city" :parameters *city-schema*
          :handler (lambda (arguments) (push arguments calls) "18°C, cloudy")))
  (with-agent-loop
    (list (stack-ai-agent:agent-run-text
           (stack-ai-agent:run-ai-agent agent "What's the weather in Paris?"))
          calls)))
#+end_src

#+RESULTS:
: ("It is 18°C, cloudy in Paris." ("{\"city\":\"Paris\"}"))

An MCP server, FastMCP's ~@mcp.tool()~, and a client in the same image:
#+begin_src lisp
(defvar *mcp-client*
  (let ((server (make-instance 'mcp-protocol:mcp-server :name "weather" :version "0.1.0"))
        (transport (rpc-backend-inprocess:make-inprocess-rpc-transport)))
    (mcp-protocol:register-tool
     server (mcp-protocol:make-mcp-tool "get_weather" :description "Current weather"
                                        :input-schema *city-schema*
                                        :handler (lambda (args) (declare (ignore args)) "21°C, sunny")))
    (mcp-protocol:serve-mcp server :transport transport)
    (let ((client (make-instance 'mcp-protocol:mcp-client :transport transport)))
      (mcp-protocol:mcp-initialize client)
      client)))

(list (mapcar #'mcp-protocol:mcp-tool-name (mcp-protocol:list-tools *mcp-client*))
      (gethash "text" (aref (gethash "content" (mcp-protocol:call-tool *mcp-client* "get_weather"
                                                                       (mcp-protocol:json-object "city" "Lyon")))
                            0)))
#+end_src

#+RESULTS:
: (("get_weather") "21°C, sunny")

RAG, LlamaIndex's ~VectorStoreIndex.from_documents(docs).as_retriever()~. The store fuses BM25 with the (mock) embeddings:
#+begin_src lisp
(let ((pipe (stack-rag:make-rag-pipeline
             :store (rag-backend-hybrid:make-hybrid-store
                     :vector-store (rag-backend-memory:make-memory-vector-store))
             :chunker (rag-backend-text:make-recursive-character-chunker :size 200 :overlap 0)
             :embedder *llm*)))
  (stack-rag:ingest pipe
                    (list (stack-rag:make-rag-document :id "lisp" :text "Common Lisp handles errors with conditions and restarts.")
                          (stack-rag:make-rag-document :id "py" :text "Python uses exceptions and try/except blocks.")
                          (stack-rag:make-rag-document :id "tea" :text "Green tea is brewed at 80 degrees.")))
  (mapcar (lambda (hit) (stack-rag:rag-chunk-text (stack-rag:rag-hit-chunk hit)))
          (stack-rag:retrieve pipe "restarts in Lisp" :top-k 1)))
#+end_src

#+RESULTS:
: ("Common Lisp handles errors with conditions and restarts.")

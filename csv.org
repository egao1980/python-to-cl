* csv
:PROPERTIES:
:UNNUMBERED: t
:END:
https://docs.python.org/3/library/csv.html

The csv module implements classes to read and write tabular data in CSV format.

** Features
:PROPERTIES:
:UNNUMBERED: t
:END:
+ ~reader~ / ~writer~: rows as lists
+ ~DictReader~ / ~DictWriter~: header rows as dicts
+ Dialects (~excel~, ~excel-tab~, ~unix~), custom delimiters, quoting modes
+ ~Sniffer~: guess the dialect and whether there is a header
+ Row by row, so large files stream

*** cl-stack
https://github.com/egao1980/cl-stack/blob/main/docs/cookbooks/csv.md

#+begin_src lisp :exports source :results silent
(asdf:load-system "csv-protocol")
(asdf:load-system "sql-query-csv")

(with-open-file (s "people.csv" :direction :output :if-exists :supersede)
  (write-string "name,city,age
Alice,Paris,31
Bob,\"Rome, IT\",25
Carol,Paris,42
" s))
#+end_src

~DictReader~:
#+begin_src python
with open("people.csv") as f:
    [(r["name"], r["city"]) for r in csv.DictReader(f)]
#+end_src

#+begin_src lisp
(map 'list (lambda (r) (list (gethash "name" r) (gethash "city" r)))
     (stack-csv:decode #p"people.csv"))
#+end_src

#+RESULTS:
: (("Alice" "Paris") ("Bob" "Rome, IT") ("Carol" "Paris"))

~reader~, rows as vectors:
#+begin_src lisp
(stack-csv:decode #p"people.csv" :header nil)
#+end_src

#+RESULTS:
: #(#("name" "city" "age") #("Alice" "Paris" "31") #("Bob" "Rome, IT" "25") #("Carol" "Paris" "42"))

Row by row, without reading the whole file:
#+begin_src lisp
(with-open-file (f "people.csv")
  (let ((in (stack-csv:make-csv-input-stream f)))
    (loop for row = (serdes-protocol:stream-decode-value in)
          until (eq row :eof)
          sum (parse-integer (gethash "age" row)))))
#+end_src

#+RESULTS:
: 98

~writer~ with a dialect; fields that need it are quoted:
#+begin_src python
csv.writer(sys.stdout, delimiter=";").writerows([["name", "note"], ["Bob", 'say "hi"']])
#+end_src

#+begin_src lisp
(stack-csv:encode '(#("name" "note") #("Bob" "say \"hi\"")) :header nil :dialect :excel-eu)
#+end_src

#+RESULTS:
#+begin_example
name;note
Bob;"say ""hi"""
#+end_example

~DictWriter~:
#+begin_src lisp
(stack-csv:encode (list (alexandria:plist-hash-table '("name" "Ann" "age" "7") :test #'equal))
                  :header '("name" "age"))
#+end_src

#+RESULTS:
#+begin_example
name,age
Ann,7
#+end_example

SQL over the file:
#+begin_src lisp
(sql-query-csv:query-csv
 (sql-query:select (sql-query:columns :city (sql-query:label (sql-query:count :*) :n))
                   (sql-query:from :people)
                   (sql-query:group-by :city)
                   (sql-query:order-by :city))
 :dialect (sql-query-csv:csv-catalog :people #p"people.csv"))
#+end_src

#+RESULTS:
: ((:CITY "Paris" :N 2) (:CITY "Rome, IT" :N 1))
